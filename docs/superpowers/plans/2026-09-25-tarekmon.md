# TarekMon Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build TarekMon, a complete single-player Pokémon-style web game (one town, one gym, one Champion) where 74 "dumpling face" WhatsApp stickers are the monsters, shipped as a static site at `tarekmon.othman.ca`.

**Architecture:** Plain ES modules with no bundler. A tiny engine (loop, canvas, input, audio, save, rng) drives a scene stack (title, overworld, battle, menus, credits). All game rules live in pure, headless modules under `src/battle/` and `src/world/` that are unit-tested with `node --test`; rendering modules under `src/ui/` and `src/world/` only draw. Content (species, moves, types, trainers, maps, dialogue, items) is JSON under `data/`.

**Tech Stack:** JavaScript (ES2022 modules), Canvas 2D, Web Audio API, `localStorage`, Node 24 `node:test` for tests, nginx Docker image for hosting on Coolify.

**Spec:** `claude-brain/designs/tarekmon-design.md` (Otto's brain repo). Read it first. Every gag and number in this plan comes from there.

**Plan layout:** this file holds the conventions, the file structure and Stages 1–2. Stages 3–4 continue in `2026-09-25-tarekmon-part2.md` in this folder. Task numbering is continuous across both files.

## Conventions for every task

- **Run tests:** `node --test` from the repo root (Node discovers `test/**/*.test.js`). Run a single file with `node --test test/battle/damage.test.js`.
- **Imports:** relative ESM paths with the `.js` extension (`import { x } from '../engine/rng.js'`). No bare specifiers, no bundler.
- **Pure vs. draw:** anything under `src/battle/`, `src/engine/save.js`, `src/engine/rng.js`, `src/world/map.js`, `src/world/encounters.js`, `src/data/` must not touch `document`, `window`, `canvas` or `localStorage` directly (inject a storage object). Everything else may.
- **Randomness:** every function that rolls dice takes an `rng` argument (a function returning `[0,1)`), defaulting to `defaultRng` from `src/engine/rng.js`. Tests pass a seeded rng.
- **Commits:** one commit per task, conventional prefix (`feat:`, `test:`, `chore:`, `content:`), message body ends with `Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>`.
- **Serve locally:** `npx serve . -l 3000` (or `python -m http.server 3000`), open `http://localhost:3000`. Verify on a phone-sized viewport (375×812) as well as desktop.
- **Copyright rule:** never copy Nintendo names, sprites, tiles or audio. The word "Pokémon" appears nowhere in `src/`, `data/` or `index.html`. Grep for it before every commit in Stage 4.
- **Stage gates with Otto:** Task 7 (content sheet) and Task 19 (look-and-feel screenshots) end with handing something to Otto and **stopping** until he answers.

```js
export function isValidType(t, types) { return types.types.includes(t); }

/** attackType may be null (typeless). Returns 2, 1 or 0.5. */
export function typeMultiplier(attackType, defenderType, types) {
  if (defenderType === types.resistsAll) return types.resistsAllMultiplier;
  if (!attackType) return 1;
  if (types.beats[attackType] === defenderType) return 2;
  if (types.beats[defenderType] === attackType) return 0.5;
  return 1;
}
```

```js
// src/data/validate.js
const EFFECTS = new Set(['none', 'accDown', 'skip', 'heal', 'atkUp', 'defUp', 'neverMiss']);

export function validateMoves(moves, types) {
  const errs = [], seen = new Set();
  for (const m of moves) {
    if (seen.has(m.id)) errs.push(`dup move id ${m.id}`); seen.add(m.id);
    if (!/^[a-z0-9-]+$/.test(m.id)) errs.push(`bad id ${m.id}`);
    if (m.type !== null && !types.types.includes(m.type)) errs.push(`${m.id}: bad type ${m.type}`);
    if (!(m.accuracy >= 1 && m.accuracy <= 100)) errs.push(`${m.id}: accuracy`);
    if (m.type === types.resistsAll && m.accuracy > 75) errs.push(`${m.id}: Imported accuracy > 75`);
    if (!(m.pp >= 5 && m.pp <= 40)) errs.push(`${m.id}: pp`);
    if (!(m.power >= 0 && m.power <= 120)) errs.push(`${m.id}: power`);
    if (!EFFECTS.has(m.effect?.kind)) errs.push(`${m.id}: effect ${m.effect?.kind}`);
    if (typeof m.text !== 'string' || !m.text) errs.push(`${m.id}: text`);
  }
  return errs;
}

export function validateSpecies(species, moves, types, stickers) {
  const errs = [], ids = new Set(), moveIds = new Set(moves.map(m => m.id)), stickerIds = new Set(stickers.map(s => s.id));
  for (const s of species) {
    if (ids.has(s.id)) errs.push(`dup species ${s.id}`); ids.add(s.id);
    if (!stickerIds.has(s.sticker)) errs.push(`${s.id}: unknown sticker ${s.sticker}`);
    if (!types.types.includes(s.type)) errs.push(`${s.id}: bad type`);
    for (const k of ['hp', 'atk', 'def', 'spd']) if (!(s.base[k] >= 20 && s.base[k] <= 130)) errs.push(`${s.id}: base ${k}`);
    if (!(s.moves.length >= 2 && s.moves.length <= 4)) errs.push(`${s.id}: ${s.moves.length} moves`);
    if (s.moves[0]?.level !== 1) errs.push(`${s.id}: first move must be level 1`);
    for (const lm of s.moves) if (!moveIds.has(lm.id)) errs.push(`${s.id}: unknown move ${lm.id}`);
    if (!(s.catchRate >= 25 && s.catchRate <= 255)) errs.push(`${s.id}: catchRate`);
    if (!(s.xpYield >= 40 && s.xpYield <= 250)) errs.push(`${s.id}: xpYield`);
    if (!s.entry || s.entry.length > 120) errs.push(`${s.id}: entry length`);
    if (!s.name || s.name.length > 10) errs.push(`${s.id}: name > 10 chars (GB text box)`);
  }
  return errs;
}
```

```js
const typeless = moves.filter(m => m.type === null);
const byType = t => moves.filter(m => m.type === t);

function pickMoves(type, id) {
  const h = hash(id);
  const own = byType(type);
  const attacks = own.filter(m => m.power > 0), status = own.filter(m => m.power === 0);
  const first = attacks[h % attacks.length];
  const second = typeless[(h >> 3) % typeless.length];
  const third = status.length ? status[(h >> 6) % status.length] : attacks[(h >> 6) % attacks.length];
  let fourth = attacks[(h >> 9) % attacks.length]; if (fourth.id === first.id) fourth = attacks[(h >> 9) % attacks.length === attacks.length - 1 ? 0 : ((h >> 9) % attacks.length) + 1];
  return [{ level: 1, id: first.id }, { level: 1, id: second.id }, { level: 8, id: third.id }, { level: 15, id: fourth.id }];
}

  const nudge = (k, i) => Math.max(20, Math.min(130, t[k] + r.bonus + ((h >> (i * 4)) % 15) - 7));
  return { id: s.id, num: i + 1, name: n.name, who: s.who, type: n.type, sticker: s.id, animated: s.animated,

    moves: pickMoves(n.type, s.id), catchRate: r.catchRate, xpYield: r.xpYield, entry: n.entry, rarity: n.rarity };
```

```js
// src/engine/rng.js
export function mulberry32(seed) {
  let a = seed >>> 0;
  return function () {
    a = (a + 0x6D2B79F5) >>> 0;
    let t = a; t = Math.imul(t ^ (t >>> 15), t | 1); t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
export const defaultRng = mulberry32((Date.now() ^ 0x9E3779B9) >>> 0);

export function dateSeed(dateString) {
  let h = 2166136261;
  for (const c of dateString) { h ^= c.charCodeAt(0); h = Math.imul(h, 16777619); }
  return h >>> 0;
}
export function todayString(d = new Date()) {
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
}
/** Nice weather 60% of calendar days, deterministic per day. */
export function rollWeather(dateString = todayString()) { return mulberry32(dateSeed(dateString))() < 0.6; }
export function randInt(rng, min, max) { return min + Math.floor(rng() * (max - min + 1)); }
export function chance(rng, p) { return rng() < p; }
export function pickWeighted(rng, items) {
  const total = items.reduce((s, i) => s + i.weight, 0);
  let r = rng() * total;
  for (const i of items) { r -= i.weight; if (r < 0) return i; }
  return items[items.length - 1];
}
```

```js
// src/engine/loop.js
export class SceneStack {
  constructor() { this.scenes = []; }
  get top() { return this.scenes[this.scenes.length - 1]; }
  push(s) { this.scenes.push(s); s.enter?.(); }
  pop() { const s = this.scenes.pop(); s?.exit?.(); return s; }
  replace(s) { this.pop(); this.push(s); }
  update(dt) { this.top?.update?.(dt); }
  draw(ctx) { this.top?.draw?.(ctx); }
  onButton(btn, pressed) { this.top?.onButton?.(btn, pressed); }
}

/** Fixed 60 Hz update, render every frame. `tick(dt)` and `render()` are injected. */
export function startLoop({ tick, render, raf = requestAnimationFrame, now = () => performance.now() }) {
  const STEP = 1 / 60; let acc = 0, last = now(), running = true;
  function frame() {
    if (!running) return;
    const t = now(); acc += Math.min(0.25, (t - last) / 1000); last = t;
    while (acc >= STEP) { tick(STEP); acc -= STEP; }
    render(); raf(frame);
  }
  raf(frame);
  return () => { running = false; };
}
```

```js
const KEYMAP = { ArrowUp: 'up', ArrowDown: 'down', ArrowLeft: 'left', ArrowRight: 'right', z: 'a', Z: 'a', x: 'b', X: 'b', Enter: 'start', ' ': 'a', Escape: 'b' };

export class ButtonState {
  constructor() { this.down = new Set(); this.pressed = new Set(); this.order = []; }
  set(btn, isDown) {
    if (isDown && !this.down.has(btn)) { this.down.add(btn); this.pressed.add(btn); this.order.push(btn); }
    if (!isDown) { this.down.delete(btn); this.order = this.order.filter(b => b !== btn); }
  }
  isDown(btn) { return this.down.has(btn); }
  justPressed(btn) { return this.pressed.has(btn); }
  direction() { for (let i = this.order.length - 1; i >= 0; i--) if (['up', 'down', 'left', 'right'].includes(this.order[i])) return this.order[i]; return null; }
```

```js
// src/world/map.js
import { WALKABLE, ENCOUNTER, INTERACT, LEGEND } from './tiles.js';
export { WALKABLE, ENCOUNTER, INTERACT };

export function loadMaps(data) {
  const out = {};
  for (const [id, m] of Object.entries(data.maps)) {
    out[id] = { id, ...m, width: m.rows[0].length, height: m.rows.length, legend: LEGEND,
      warps: m.warps ?? [], zones: m.zones ?? [], npcs: m.npcs ?? [], trainers: m.trainers ?? [], signs: m.signs ?? [], objects: m.objects ?? [] };
  }
  return out;
}
export const inBounds = (m, x, y) => x >= 0 && y >= 0 && x < m.width && y < m.height;
export const tileAt = (m, x, y) => inBounds(m, x, y) ? m.rows[y][x] : null;
export function isWalkable(m, x, y, flags = {}) {
  const t = tileAt(m, x, y); if (t === null) return false;
  if (t === 'G') return false; // closed gate; the overworld swaps G→g when the flag is set
  return WALKABLE.has(t);
}
export const isEncounterTile = (m, x, y) => ENCOUNTER.has(tileAt(m, x, y));
export const warpAt = (m, x, y) => m.warps.find(w => w.x === x && w.y === y) ?? null;
export const zoneAt = (m, x, y) => m.zones.find(z => x >= z.rect[0] && y >= z.rect[1] && x < z.rect[0] + z.rect[2] && y < z.rect[1] + z.rect[3]) ?? null;
export const npcAt = (m, x, y) => [...m.npcs, ...m.trainers].find(n => n.x === x && n.y === y) ?? null;
export const signAt = (m, x, y) => m.signs.find(s => s.x === x && s.y === y) ?? null;
export const objectAt = (m, x, y) => m.objects.find(o => o.x === x && o.y === y) ?? null;
export function step(x, y, facing) { return facing === 'up' ? [x, y - 1] : facing === 'down' ? [x, y + 1] : facing === 'left' ? [x - 1, y] : [x + 1, y]; }
```
